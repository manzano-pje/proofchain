camada servço de validação

@Service
public class PlanValidationService {

    public void validateLimit(UUID institutionId, String featureKey) {

        Subscription subscription = subscriptionRepository.findActiveByInstitution(institutionId);
        if(subscription == null || subscription.getExpiresAt().isBefore(LocalDateTime.now())) {
            throw new BusinessException("Plano expirado.");
        }

        PlanFeature feature = planFeatureRepository
                .findByPlanAndFeatureKey(subscription.getPlan().getId(), featureKey);

        if(feature == null || !feature.isEnabled()) {
            throw new BusinessException("Recurso não disponível no seu plano.");
        }

        if(feature.getLimitValue() != null) {

            int currentUsage = usageService.getUsage(institutionId, featureKey);

            if(currentUsage >= feature.getLimitValue()) {
                throw new BusinessException("Limite atingido para o recurso: " + featureKey);
            }
        }
    }
}
